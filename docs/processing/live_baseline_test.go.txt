package generator
import("context";"testing";"os";"time";"encoding/json";"sync";"log/slog";"github.com/mgrubiyan/pomnibot/backend/internal/ingest";"github.com/mgrubiyan/pomnibot/backend/internal/providers";"github.com/mgrubiyan/pomnibot/backend/internal/providers/openrouter")
type baselineTracker struct {p providers.Provider;mu sync.Mutex;Calls,PromptTokens,CompletionTokens int;CostUSD float64;Models map[string]int}
func(b *baselineTracker)Complete(ctx context.Context,r providers.Request)(providers.Response,error){resp,err:=b.p.Complete(ctx,r);b.mu.Lock();defer b.mu.Unlock();b.Calls++;b.PromptTokens+=resp.Usage.PromptTokens;b.CompletionTokens+=resp.Usage.CompletionTokens;b.CostUSD+=resp.Usage.CostUSD;b.Models[resp.Model]++;return resp,err}
func TestLiveBaseline(t *testing.T){
 if os.Getenv("POMNIBOT_LIVE_BASELINE")!="1" {t.Skip("explicit live opt-in required")}
 slog.SetLogLoggerLevel(slog.LevelWarn)
 cfg,err:=openrouter.ConfigFromEnv();if err!=nil{t.Fatal(err)}
 p,err:=openrouter.New(cfg);if err!=nil{t.Fatal(err)}
 tracker:=&baselineTracker{p:p,Models:map[string]int{}}
 data,err:=os.ReadFile(os.Getenv("POMNIBOT_EVAL_PDF"));if err!=nil{t.Fatal(err)}
 ctx:=context.Background();began:=time.Now()
 ex,err:=ingest.NewExtractor(&ingest.VisionOCR{Model:tracker,FullPage:true},ingest.Options{ReadAllPDFPages:true});if err!=nil{t.Fatal(err)}
 reading,err:=ex.Extract(ctx,[]ingest.File{{Name:"SQL.pdf",Data:data}});if err!=nil{t.Fatal(err)}
 readTime:=time.Since(began);first:=time.Duration(0)
 gen:=NewGenerator(tracker,Options{OnBatch:func(Batch){if first==0 {first=time.Since(began)}}})
 doc:=Document{Text:reading.Text,Title:"Введение в SQL"};for _,p:=range reading.Pages{doc.PageStarts=append(doc.PageStarts,p.Start)}
 result,genErr:=gen.Generate(ctx,doc)
 out:=struct{Result Result `json:"result"`;Reading ingest.Result `json:"reading"`;Model string `json:"model"`;Calls int `json:"calls"`;PromptTokens int `json:"promptTokens"`;CompletionTokens int `json:"completionTokens"`;CostUSD float64 `json:"costUsd"`;Models map[string]int `json:"models"`;ReadSeconds,FirstCardSeconds,ElapsedSeconds float64;Error string `json:"error,omitempty"`}{Result:result,Reading:reading,Model:cfg.Model,Calls:tracker.Calls,PromptTokens:tracker.PromptTokens,CompletionTokens:tracker.CompletionTokens,CostUSD:tracker.CostUSD,Models:tracker.Models,ReadSeconds:readTime.Seconds(),FirstCardSeconds:first.Seconds(),ElapsedSeconds:time.Since(began).Seconds()}
 if genErr!=nil{out.Error=genErr.Error()};b,err:=json.MarshalIndent(out,"","  ");if err!=nil{t.Fatal(err)}
 if err=os.WriteFile(os.Getenv("POMNIBOT_BASELINE_OUTPUT"),b,0600);err!=nil{t.Fatal(err)}
 t.Logf("baseline complete: cards=%d calls=%d cost=$%.6f took=%.1fs",len(result.Cards),tracker.Calls,tracker.CostUSD,time.Since(began).Seconds())
 if genErr!=nil{t.Fatal(genErr)}
}
